1. 공통코드 조회 및 동적 드롭다운 콤보 바인딩

Public Function CodeList_Select(sKeyID, [sParameter]) As String Public Function

DB 저장 프로시저 dbo.Code_Select를 호출하여 엑셀 유효성 검사에 즉시 주입 가능한 콤마(,) 구분 문자열을 반환합니다.

Public Function CodeList_Select(ByRef sKeyID As String, Optional sParameter As String = "") As String
    Dim xResult As Variant
    xResult = ExecCmd("dbo.Code_Select '" & sKeyID & "', '" & sParameter & "'")
    
    If IsNull(xResult(0, 0)) Then
        CodeList_Select = ""
    Else
        CodeList_Select = xResult(0, 0)
    End If
End Function
Public Function RangeCode_Set(xTarget, sAddress, [bFirstValueDisplay]) As Boolean Public Function

대상 셀 범위(xTarget)에 엑셀 드롭다운 목록(xlValidateList)을 동적으로 등록하고, 옵션에 따라 첫 번째 항목을 자동 선택합니다.

Public Function RangeCode_Set(xTarget As Range, sAddress As String, Optional bFirstValueDisplay As Boolean = False) As Boolean
    Application.ScreenUpdating = False
    
    If sAddress <> "" Then
        With xTarget.Validation
            .Delete
            .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:=sAddress
        End With
        RangeCode_Set = True
    Else
        xTarget.Validation.Delete
        RangeCode_Set = False
    End If

    If bFirstValueDisplay Then
        xTarget.value = ComboFirst(sAddress)
    End If
    Application.ScreenUpdating = True
End Function

2. 변경 영역 검증 및 필수 입력값 무결성 체크

Public Function Change_Check(Target, xDataRange) As Boolean Public Function

Worksheet_Change 이벤트에서 수정된 셀이 실제 비즈니스 데이터 입력 영역(Data Range) 내부에 속하는지 경계값을 판별합니다.

Public Function Change_Check(ByRef Target As Range, ByRef xDataRange As Range) As Boolean
    If Target.Row < xDataRange.Row Then Change_Check = False: Exit Function
    If Target.Row >= xDataRange.Row + xDataRange.Rows.Count Then Change_Check = False: Exit Function
    If Target.Column < xDataRange.Column Then Change_Check = False: Exit Function
    If Target.Column >= xDataRange.Column + xDataRange.Columns.Count Then Change_Check = False: Exit Function
    Change_Check = True
End Function
Public Function AllDataInput_Check(xTarget As Range) As Boolean Public Function

단일 행에 대해 공백 셀이 하나라도 존재하는지 순회하여, 저장 전 필수 입력값 누락을 방지합니다.
