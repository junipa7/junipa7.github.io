mod_01_DbConnect.bas 
' ========================================================
' [MES20230221] MS-SQL ADODB 연결 및 트랜잭션 관리 모듈
' ========================================================
Public DB_Conn As ADODB.Connection
Public DB_Rs As ADODB.Recordset

Public Function ConnectDB() As Boolean
    On Error GoTo ErrHandler
    Set DB_Conn = New ADODB.Connection
    
    Dim strConn As String
    strConn = "Provider=SQLOLEDB;Data Source=192.168.0.10,1433;" & _
              "Initial Catalog=SMART_MES_DB;" & _
              "User ID=mes_admin;Password=mes_secure_pwd;"
              
    DB_Conn.Open strConn
    ConnectDB = True
    Exit Function
    
ErrHandler:
    MsgBox "DB 접속 실패: " & Err.Description, vbCritical, "MES 연결 오류"
    ConnectDB = False
End Function

Public Sub DisconnectDB()
    If Not DB_Rs Is Nothing Then
        If DB_Rs.State = adStateOpen Then DB_Rs.Close
        Set DB_Rs = Nothing
    End If
    If Not DB_Conn Is Nothing Then
        If DB_Conn.State = adStateOpen Then DB_Conn.Close
        Set DB_Conn = Nothing
    End If
End Sub
